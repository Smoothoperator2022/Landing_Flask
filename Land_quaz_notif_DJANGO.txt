def demo_form(request):
    if request.method == 'POST':
        form = DemoRequestForm(request.POST)
        if form.is_valid():
            name = form.cleaned_data['name']
            phone = form.cleaned_data['phone']
            email = form.cleaned_data['email']

            # >>> ВКАЖІТЬ СВОЇ АДРЕСИ ОТРИМУВАЧІВ <<<
            recipients = ["first@example.com", "second@example.com"]

            subject = "Запит на демонстрацію з лендингу"
            body = (
                f"Ім'я: {name}\n"
                f"Телефон: {phone}\n"
                f"E-mail: {email}\n"
            )

            # Варіант 1: простий send_mail
            send_mail(
                subject,
                body,
                settings.DEFAULT_FROM_EMAIL,  # від кого
                recipients,
                fail_silently=False
            )

            # Або Варіант 2 (коментуйте/розкоментовуйте за потреби):
            # msg = EmailMessage(subject, body, settings.DEFAULT_FROM_EMAIL, recipients)
            # msg.reply_to = [email] if email else []
            # msg.send(fail_silently=False)

            messages.success(request, "Дякуємо! Заявку надіслано.")
            return redirect('landing:demo')  # повернемось на форму з повідомленням
    else:
        form = DemoRequestForm()

    return render(request, 'landing/demo_form.html', {'form': form})

4) Форма (валідація)
landing/forms.py
from django import forms

class DemoRequestForm(forms.Form):
    name = forms.CharField(
        label="Ім'я",
        max_length=100,
        widget=forms.TextInput(attrs={"placeholder": "Ваше ім'я", "class": "input"})
    )
    phone = forms.CharField(
        label="Телефон",
        max_length=30,
        widget=forms.TextInput(attrs={"placeholder": "+49 123 4567", "class": "input"})
    )
    email = forms.EmailField(
        label="Електронна пошта",
        widget=forms.EmailInput(attrs={"placeholder": "you@example.com", "class": "input"})
    )

6) Налаштування email

У settings.py (поставте реальні SMTP значення). Для швидкої перевірки локально можна використовувати console backend (пошта друкується в консолі), а далі — бойовий SMTP.

Варіант A: консольний (для тесту)
EMAIL_BACKEND = "django.core.mail.backends.console.EmailBackend"
DEFAULT_FROM_EMAIL = "no-reply@example.com"

Варіант B: реальний SMTP (наприклад, Gmail або ваш SMTP)
EMAIL_BACKEND = "django.core.mail.backends.smtp.EmailBackend"
EMAIL_HOST = "smtp.gmail.com"
EMAIL_PORT = 587
EMAIL_HOST_USER = "your_account@gmail.com"
EMAIL_HOST_PASSWORD = "your_app_password"  # саме app password
EMAIL_USE_TLS = True
DEFAULT_FROM_EMAIL = EMAIL_HOST_USER

7) Запуск
python manage.py makemigrations
python manage.py migrate
python manage.py runserver


Сторінка №1: http://127.0.0.1:8000/ — кнопка «Запустить демонстрацию» (єдина клікабельна).

Сторінка №2: http://127.0.0.1:8000/demo/ — форма з полями Ім’я / Телефон / Електронна пошта.

Після відправки — лист іде на дві адреси, а сторінка показує зелене повідомлення «Дякуємо! Заявку надіслано.»
